public class AccountTierBatch implements

 return Database.getQueryLocator(

    FROM Account 
    WHERE status = 'Active'
    WITH USER_MODE ];
 )



}

public void execute(Database.BatchableContext bc ,List<Account> acc){
      String Annual_Revenue;
      String newtier;
      Integer total =0;
      Integer ent=0;
      Integer med =0;
      Integer small=0;
   List<Account> accounts = new List<Account>();
   for(Account a : acc ){
    if(a.AnnualRevenue ≥ 10,000,000){
        a.Customer_Tier__c='Enterprise';
        ent++;
    }
    elseif((a.AnnualRevenue ≥ 1,000,000)
    && (a.AnnualRevenue <10,000,000) ){
          a.Customer_Tier__c='Medium';
          med++;

    }
    elseif(a.AnnualRevenue << 1,000,000){
        a.Customer_Tier__c = 'Small';
        Small++;
    }
    else {
       return 0;
    }
    accounts.add(a);
    total ++;

   }
   Database.update (accounts,false);
}
