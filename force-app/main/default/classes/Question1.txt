public with sharing class getAccount{
         public static List<Account> getRecordServiceAccount(Id accId){
            return [SELECT NAME ,
                     (SELECT Id,Name,Email,Phone From Contacts),
                     (SELECT Id,Name,StageName,Amount,CloseDate FROM Opportunities),
                     WHERE AccountId IN : accId
                     WITH USER_MODE
                     ];

                      




         }
}
